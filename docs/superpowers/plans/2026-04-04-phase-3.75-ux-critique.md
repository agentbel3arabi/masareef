```python
# backend/app/services/transaction_summary.py
"""Transaction summary aggregation. No HTTP awareness."""

import datetime
import uuid
from dateutil.relativedelta import relativedelta

from sqlalchemy import case, func, select
from sqlalchemy.ext.asyncio import AsyncSession

from app.models.transaction import Transaction
from app.schemas.transaction_summary import TransactionSummaryData, TransactionSummaryPeriod


def _resolve_period(
    period: str,
    start_date: datetime.date | None,
    end_date: datetime.date | None,
) -> tuple[datetime.date, datetime.date]:
    """Resolve period string to concrete start/end dates."""
    today = datetime.date.today()
    if period == "custom":
        if start_date is None or end_date is None:
            raise ValueError("start_date and end_date required for custom period")
        return start_date, end_date
    if period == "month":
        start = today.replace(day=1)
        end = (start + relativedelta(months=1)) - datetime.timedelta(days=1)
        return start, end
    if period == "quarter":
        q_start_month = ((today.month - 1) // 3) * 3 + 1
        start = today.replace(month=q_start_month, day=1)
        end = (start + relativedelta(months=3)) - datetime.timedelta(days=1)
        return start, end
    if period == "year":
        start = today.replace(month=1, day=1)
        end = today.replace(month=12, day=31)
        return start, end
    raise ValueError(f"Invalid period: {period}")


async def get_transaction_summary(
    session: AsyncSession,
    household_id: uuid.UUID,
    *,
    period: str = "month",
    start_date: datetime.date | None = None,
    end_date: datetime.date | None = None,
    account_id: int | None = None,
    category_id: int | None = None,
    type: str | None = None,
) -> TransactionSummaryData:
    """Aggregate transaction totals for a period with optional filters."""
    p_start, p_end = _resolve_period(period, start_date, end_date)

    # Build base query — only active transactions in household within period
    conditions = [
        Transaction.household_id == household_id,
        Transaction.is_active.is_(True),
        Transaction.date >= p_start,
        Transaction.date <= p_end,
    ]

    if account_id is not None:
        conditions.append(Transaction.account_id == account_id)
    if category_id is not None:
        conditions.append(Transaction.category_id == category_id)
    if type == "income":
        conditions.append(Transaction.amount_minor > 0)
    elif type == "expense":
        conditions.append(Transaction.amount_minor < 0)

    stmt = select(
        func.coalesce(
            func.sum(case((Transaction.amount_minor > 0, Transaction.amount_minor), else_=0)),
            0,
        ).label("total_income"),
        func.coalesce(
            func.sum(case((Transaction.amount_minor < 0, func.abs(Transaction.amount_minor)), else_=0)),
            0,
        ).label("total_expenses"),
        func.count().label("tx_count"),
    ).where(*conditions)

    result = await session.execute(stmt)
    row = result.one()

    total_income = int(row.total_income)
    total_expenses = int(row.total_expenses)

    return TransactionSummaryData(
        total_income=total_income,
        total_expenses=total_expenses,
        net_flow=total_income - total_expenses,
        transaction_count=int(row.tx_count),
        currency="EGP",  # Primary currency; multi-currency deferred to Phase 4
        period=TransactionSummaryPeriod(start=p_start, end=p_end),
    )
```

```python
async def get_balance_history(
    session: AsyncSession,
    household_id: uuid.UUID,
    account_id: int,
    period: str = "month",
) -> dict:
    """Compute balance change for an account over a period."""
    account = await get_account(session, household_id, account_id)
    if account is None:
        raise ValueError(f"Account {account_id} not found")

    today = datetime.date.today()
    period_start = today.replace(day=1)  # Start of current month

    # Sum all transactions in the period
    from app.models.transaction import Transaction

    stmt = select(func.coalesce(func.sum(Transaction.amount_minor), 0)).where(
        Transaction.account_id == account_id,
        Transaction.household_id == household_id,
        Transaction.is_active.is_(True),
        Transaction.date >= period_start,
    )
    result = await session.execute(stmt)
    period_sum = int(result.scalar())

    current = account.displayed_balance_minor
    period_start_balance = current - period_sum
    change = current - period_start_balance

    if change > 0:
        direction = "up"
    elif change < 0:
        direction = "down"
    else:
        direction = "unchanged"

    return {
        "current_balance": current,
        "period_start_balance": period_start_balance,
        "change": change,
        "change_direction": direction,
        "period": {"start": str(period_start), "end": str(today)},
    }
```

```tsx
  secondary?: ReactNode;

            />
            <DropdownMenuContent align="end">
              {secondaryMenuItems}
            </DropdownMenuContent>
          </DropdownMenu>
        </div>
```

### Task 7.1: Transaction summary hook

- [ ] **Step 1: Create the hook**

```tsx
// frontend/src/hooks/use-transaction-summary.ts
"use client";

import { useQuery } from "@tanstack/react-query";
import { apiGet } from "@/lib/api-client";

interface TransactionSummary {
  total_income: number;
  total_expenses: number;
  net_flow: number;
  transaction_count: number;
  currency: string;
  period: { start: string; end: string };
}

interface SummaryParams {
  period?: string;
  start_date?: string;
  end_date?: string;
  account_id?: number;
}

export function useTransactionSummary(params: SummaryParams = {}) {
  const searchParams = new URLSearchParams();
  if (params.period) searchParams.set("period", params.period);
  if (params.start_date) searchParams.set("start_date", params.start_date);
  if (params.end_date) searchParams.set("end_date", params.end_date);
  if (params.account_id) searchParams.set("account_id", String(params.account_id));
  const qs = searchParams.toString();

  return useQuery({
    queryKey: ["transaction-summary", params],
    queryFn: () => apiGet<TransactionSummary>(`/api/v1/transactions/summary${qs ? `?${qs}` : ""}`),
  });
}
```

```tsx
import { useState, useEffect } from "react";

import { useTranslations, useLocale } from "next-intl";

  Wallet, Receipt, ArrowLeftRight, HandCoins, LayoutDashboard,

  CommandGroup,
  CommandInput,

  CommandList,

import { useAccounts } from "@/hooks/use-accounts";
import { formatAmount, formatAmountAr } from "@/lib/money";
import { navItems } from "@/lib/nav-items";

export function CommandPalette() {
  const [open, setOpen] = useState(false);
  const router = useRouter();
  const t = useTranslations();
  const locale = useLocale();

  const { data: accountsData } = useAccounts();
  const accounts = accountsData?.data ?? [];

  // Global keyboard shortcut
  useEffect(() => {
    const handler = (e: KeyboardEvent) => {
      if ((e.metaKey || e.ctrlKey) && e.key === "k") {
        e.preventDefault();
        setOpen((prev) => !prev);
      }
    };
    document.addEventListener("keydown", handler);
    return () => document.removeEventListener("keydown", handler);
  }, []);

  const navigate = (href: string) => {
    router.push(href);
    setOpen(false);
  };

  const fmt = (amount: number, currency: string) =>
    locale === "ar" ? formatAmountAr(amount, currency) : formatAmount(amount, currency);

  const activeNavItems = navItems.filter((i) => !i.disabled);

  return (
    <CommandDialog open={open} onOpenChange={setOpen}>
      <CommandInput placeholder={t("search.placeholder")} />
      <CommandList>
        <CommandEmpty>{t("search.noResults")}</CommandEmpty>

        <CommandGroup heading={t("search.pages")}>
          {activeNavItems.map((item) => (
            <CommandItem key={item.href} onSelect={() => navigate(item.href)}>
              <item.icon className="me-2 h-4 w-4" />
              {t(item.label)}
            </CommandItem>
          ))}
        </CommandGroup>

        {accounts.length > 0 && (
          <CommandGroup heading={t("search.accounts")}>
            {accounts.slice(0, 5).map((acct) => (
              <CommandItem key={acct.id} onSelect={() => navigate(`/accounts/${acct.id}`)}>

                  {fmt(acct.displayed_balance_minor, acct.currency)}
                </span>
              </CommandItem>
            ))}
          </CommandGroup>

  );
}
```

- [ ] **Step 2: Create NavbarActions component**

```tsx
// frontend/src/components/layout/navbar-actions-portal.tsx
"use client";

import { useEffect, type ReactNode } from "react";
import { useNavbarActions } from "@/contexts/navbar-actions-context";

interface NavbarActionsProps {
  children: ReactNode;
}

/**
 * Declarative component for setting navbar actions.
 * Pages render this in their JSX — no useEffect needed.
 *
 * Usage:
 *   <NavbarActions>
 *     <Button>My Action</Button>
 *   </NavbarActions>
 */
export function NavbarActions({ children }: NavbarActionsProps) {
  const { setActions } = useNavbarActions();

  useEffect(() => {
    setActions(children);
    return () => setActions(null);
  }, [children, setActions]);

  return null;
}
```

### Task 9.2: Bulk action hook (Finding 7.2)

- [ ] **Step 1: Extract shared bulk action logic**

The `useBulkSelection` hook already exists at `frontend/src/hooks/use-bulk-selection.ts`. The remaining duplication is the navbar action rendering for manage mode. Create a hook that combines bulk selection with navbar actions:

```tsx
// frontend/src/hooks/use-bulk-action-navbar.tsx
"use client";

import { type ReactNode } from "react";
import { useTranslations } from "next-intl";
import { Trash2, Settings } from "lucide-react";
import { useBulkSelection } from "@/hooks/use-bulk-selection";
import { Button } from "@/components/ui/button";

interface UseBulkActionNavbarOptions {
  onDelete: (ids: Set<number | string>) => Promise<void>;
  deleteIsPending: boolean;
  entityLabel: string;
  /** Additional actions to show in non-manage mode */
  normalActions?: ReactNode;
}

export function useBulkActionNavbar(options: UseBulkActionNavbarOptions) {
  const t = useTranslations();
  const {
    bulkMode,
    selectedIds,
    enterBulkMode,
    exitBulkMode,
    toggleSelect,
    selectAll,
  } = useBulkSelection();

  const handleDelete = async () => {
    await options.onDelete(selectedIds);
    exitBulkMode();
  };

  let navbarContent: ReactNode;

  if (!bulkMode) {
    navbarContent = (
      <div className="flex items-center gap-2">
        <Button variant="outline" size="sm" onClick={enterBulkMode}>
          <Settings className="h-4 w-4 me-1" />
          {t("transactions.manage")}
        </Button>
        {options.normalActions}
      </div>
    );
  } else if (selectedIds.size === 0) {
    navbarContent = (
      <Button variant="secondary" size="sm" onClick={exitBulkMode}>
        {t("common.cancel")}
      </Button>
    );
  } else {
    navbarContent = (
      <div className="flex items-center gap-2">
        <span className="text-sm text-muted-foreground">
          {selectedIds.size} {options.entityLabel}
        </span>
        <Button
          variant="destructive"
          size="sm"
          disabled={options.deleteIsPending}
          onClick={handleDelete}
        >
          <Trash2 className="h-3.5 w-3.5 me-1" />
          {t("common.delete.button")} ({selectedIds.size})
        </Button>
        <Button variant="ghost" size="sm" onClick={exitBulkMode}>
          {t("common.cancel")}
        </Button>
      </div>
    );
  }

  return {
    bulkMode,
    selectedIds,
    toggleSelect,
    selectAll,
    navbarContent,
    enterBulkMode,
    exitBulkMode,
  };
}
```
